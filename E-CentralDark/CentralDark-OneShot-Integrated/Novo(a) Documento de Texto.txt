<html lang="pt-BR">
<head>
  <meta charset="UTF-8">
  <title>Painel Espelhado</title>
  <style>
    #progressBar {
      width: 100%;
      background-color: #ddd;
    }
    #progress {
      width: 0%;
      height: 30px;
      background-color: #4caf50;
      text-align: center;
      color: white;
    }
  </style>
</head>
<body>
  <h2>Painel de Tokens Espelhado</h2>
  <textarea id="inputText" rows="5" cols="50" placeholder="Digite seu texto aqui..."></textarea>
  <div id="progressBar">
    <div id="progress">0%</div>
  </div>
  <p id="stats"></p>

  <script>
    const limiteTokens = 1000;
    const maxCaracteres = limiteTokens * 4;

    const input = document.getElementById("inputText");
    const progress = document.getElementById("progress");
    const stats = document.getElementById("stats");

    input.addEventListener("input", () => {
      const texto = input.value;
      const letras = texto.replace(/\s/g, "").length;
      const palavras = texto.trim().split(/\s+/).filter(Boolean).length;

      const tokensUsados = Math.ceil(letras / 4);
      const tokensRestantes = limiteTokens - tokensUsados;
      const percentual = Math.min(100, Math.round((tokensUsados / limiteTokens) * 100));

      progress.style.width = percentual + "%";
      progress.textContent = percentual + "%";

      stats.textContent = `
        Letras usadas: ${letras} | Restantes: ${maxCaracteres - letras}
        Palavras usadas: ${palavras}
        Tokens usados: ${tokensUsados} | Restantes: ${tokensRestantes}
      `;
    });
  </script>
</body>
</html>
