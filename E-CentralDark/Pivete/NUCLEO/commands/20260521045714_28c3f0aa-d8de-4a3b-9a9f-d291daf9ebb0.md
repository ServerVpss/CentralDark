  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  workspace_id uuid NULL,

CREATE OR REPLACE FUNCTION public.consume_rate_limit(
  _action text,
  _limit integer,
  _window_seconds integer,
  _actor_user_id uuid DEFAULT NULL,
  _workspace_id uuid DEFAULT NULL,
  _ip_hash text DEFAULT NULL
) RETURNS TABLE(allowed boolean, retry_after_seconds integer, current_count integer)
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
  v_row public.rate_limits%ROWTYPE;
  v_now timestamptz := now();
  v_window_start timestamptz;
BEGIN
  v_window_start := v_now - make_interval(secs => _window_seconds);

  SELECT * INTO v_row FROM public.rate_limits
   WHERE action = _action
     AND COALESCE(actor_user_id::text,'') = COALESCE(_actor_user_id::text,'')
     AND COALESCE(workspace_id::text,'')  = COALESCE(_workspace_id::text,'')
     AND COALESCE(ip_hash,'')             = COALESCE(_ip_hash,'')
   FOR UPDATE;

  IF NOT FOUND THEN
    INSERT INTO public.rate_limits (action, actor_user_id, workspace_id, ip_hash, count, limit_value, window_start)
    VALUES (_action, _actor_user_id, _workspace_id, _ip_hash, 1, _limit, v_now);
    RETURN QUERY SELECT true, 0, 1;
    RETURN;
  END IF;

  IF v_row.window_start < v_window_start THEN
    UPDATE public.rate_limits SET count = 1, window_start = v_now, limit_value = _limit, blocked_until = NULL, updated_at = v_now
     WHERE id = v_row.id;
    RETURN QUERY SELECT true, 0, 1;
    RETURN;
  END IF;

  IF v_row.count >= _limit THEN
    UPDATE public.rate_limits SET blocked_until = v_row.window_start + make_interval(secs => _window_seconds), updated_at = v_now
     WHERE id = v_row.id;
    RETURN QUERY SELECT
      false,
      GREATEST(1, EXTRACT(EPOCH FROM ((v_row.window_start + make_interval(secs => _window_seconds)) - v_now))::int),
      v_row.count;
    RETURN;
  END IF;

  UPDATE public.rate_limits SET count = v_row.count + 1, updated_at = v_now, limit_value = _limit
   WHERE id = v_row.id;
  RETURN QUERY SELECT true, 0, v_row.count + 1;
END $$;

LANGUAGE plpgsql

DECLARE
  new_workspace_id UUID;
  base_slug TEXT;
  final_slug TEXT;
  attempt INT := 0;
  v_skip BOOLEAN;
  v_signup_enabled BOOLEAN;
BEGIN
  INSERT INTO public.profiles (id, email, full_name)
  VALUES (
    NEW.id,
    NEW.email,
    COALESCE(NEW.raw_user_meta_data->>'full_name', split_part(NEW.email, '@', 1))
  )
  ON CONFLICT (id) DO NOTHING;

  v_skip := COALESCE((NEW.raw_user_meta_data->>'skip_default_workspace')::boolean, false);
  IF v_skip THEN
    RETURN NEW;
  END IF;

  SELECT COALESCE(public_signup_enabled, true) INTO v_signup_enabled
    FROM public.master_settings WHERE id = 1;

  IF NOT COALESCE(v_signup_enabled, true) THEN
    RAISE EXCEPTION 'Cadastro público desativado. Solicite um convite ao administrador.'
      USING ERRCODE = '42501';
  END IF;

  base_slug := lower(regexp_replace(split_part(NEW.email, '@', 1), '[^a-z0-9]+', '-', 'g'));
  IF base_slug = '' THEN base_slug := 'workspace'; END IF;
  final_slug := base_slug;
  WHILE EXISTS (SELECT 1 FROM public.workspaces WHERE slug = final_slug) LOOP
    attempt := attempt + 1;
    final_slug := base_slug || '-' || attempt::text;
  END LOOP;

  INSERT INTO public.workspaces (name, slug, owner_id)
  VALUES (
    COALESCE(NEW.raw_user_meta_data->>'workspace_name', 'Minha Empresa'),
    final_slug,
    NEW.id
  )
  RETURNING id INTO new_workspace_id;

  INSERT INTO public.workspace_members (workspace_id, user_id, role)
  VALUES (new_workspace_id, NEW.id, 'admin');

  BEG
