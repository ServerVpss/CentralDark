put("03-GAPS-REAIS.md", r"""

Gaps reais — o que falta para funcionar na Casa
GAP 1 — Scheduler físico
O próprio Memory Linker declara:

SCHEDULE=UNPROVEN
O Dream Review não agenda a si mesmo.

É necessário medir qual scheduler é realmente o owner local:

A) scheduler nativo do Hermes local
B) Windows Task Scheduler
C) outro scheduler canônico já instalado
A regra é:

DISCOVER_EXISTING
→ REUSE
→ somente se nenhum owner aplicável existir:
   HUMAN_SELECTION_REQUIRED
GAP 2 — Janela noite/dia
Não existe no núcleo do Dream Review um contrato físico de:

NIGHT_START
NIGHT_STOP
WAKE_TIME
Este pack cria o contrato, mas deixa os horários REQUIRED até o CHEFE definir.

GAP 3 — Logs diurnos
O Dream Review já aceita eventos JSONL e artefatos, porém não há prova de que o log completo do Hermes local esteja convertido para o schema Dream.

Logo:

DAY_LOG_EXISTS? = UNPROVEN
DAY_LOG_SCHEMA = UNPROVEN
DAY_LOG_ADAPTER = UNPROVEN
Não parsear log arbitrário no escuro.

Primeiro medir o log real; então adaptar o menor formato necessário.

GAP 4 — Hard stop
Cron de início não garante parada.

É preciso deadline dentro do executor.

O wrapper deste pack:

calcula deadline;

monitora processo;

respeita STOP.REQUESTED;

mata somente o processo filho do ciclo se ultrapassar a janela;

grava receipt.

GAP 5 — Wake receipt
É preciso separar:

NIGHT_JOB_FINISHED
de:

DAY_HERMES_READY
wake lê o receipt e volta ao modo interativo sem promover candidatos.

GAP 6 — Conflito com trabalho diurno
Se a Casa ainda estiver em trabalho ativo quando a janela começar:

ACTIVE_INTERACTIVE_LOCK
→ DREAM=DEFER
Não roubar Writer nem executar sobre estado parcialmente mutado.

GAP 7 — Estado do clone local
O script atual de Casa/Obsidian explicitamente bloqueia defaults históricos de C:\/E:\ até os roots serem medidos.

Portanto este pack também não hardcoda root.

GAP 8 — Review-of-review
O core já tem MetaLupa e testes, mas o ciclo físico noturno precisa produzir um receipt explícito de replay.

Este pack acrescenta:

PASS1_HASH
PASS2_HASH
REPLAY_STABLE=true|false
Sem transformar replay em Judge independente.
""")

const repoRoot = path.resolve(repoRootArg);
const proposalPath = path.resolve(proposalArg);
const expectedHead = String(expectedHeadArg).trim().toLowerCase();

if (!/^[a-f0-9]{40}$/.test(expectedHead)) throw new Error('EXPECTED_HEAD_INVALID');

const dreamModule = await import(pathToFileURL(

).href);
const patternModule = await import(pathToFileURL(

).href);

const proposal = JSON.parse(fs.readFileSync(proposalPath, 'utf8'));
const dreamValidation = dreamModule.validateDreamProposal(proposal);
const patternValidation = patternModule.validatePatternDiscovery(proposal);

if (!dreamValidation.ok) errors.push(...dreamValidation.errors.map((x) => DREAM:${x}));
if (!patternValidation.ok) errors.push(...patternValidation.errors.map((x) => PATTERN:${x}));
if (String(proposal.head_sha || '').toLowerCase() !== expectedHead) errors.push('HEAD_MISMATCH');
if (proposal.apply_allowed !== false) errors.push('APPLY_NOT_FALSE');
if (proposal.direct_canonical_mutation !== false) errors.push('DIRECT_CANONICAL_MUTATION_NOT_FALSE');
if (proposal.raw_session_mutation !== false) errors.push('RAW_SESSION_MUTATION_NOT_FALSE');
if (proposal.autonomy_routing?.transport_execution === true) errors.push('TRANSPORT_EXECUTION_TRUE');

const receipt = {
schema_version: 1,
mode: 'DREAM_HOUSE_LOCAL_JUDGE',
expected_head: expectedHead,

proposal_status: proposal.status ?? null,

dream_validation: dreamValidation,
pattern_validation: patternValidation,
independent_judge: false,
self_review_is_approval: false,

errors,
};

[CmdletBinding()]
param(
[Parameter(Mandatory=$true)][string]$RepoRoot,
[Parameter(Mandatory=$true)][string]$StateRoot,
[Parameter(Mandatory=$true)][ValidatePattern('^\d{2}:\d{2}$')][string]$NightStop,
[string]$InputJsonl = "",
[switch]$DryRun
)

Set-StrictMode -Version Latest
$ErrorActionPreference = 'Stop'

function Fail([string]$Code, [string]$Message) {
$obj = [ordered]@{
timestamp = (Get-Date).ToString("o")
status = $Code
pass = $false
message = $Message
}
$obj | ConvertTo-Json -Depth 8
exit 3
}

$repo = [System.IO.Path]::GetFullPath($RepoRoot)
$state = [System.IO.Path]::GetFullPath($StateRoot)

if (-not (Test-Path $repo -PathType Container)) { Fail "BLOCKED_REPO_MISSING" $repo }
if (-not (Test-Path (Join-Path $repo ".git"))) { Fail "BLOCKED_NOT_GIT_REPO" $repo }

$required = @(

"scripts/memory-linker/dream-review-run.mjs",

Fail "BLOCKED_DREAM_FILE_MISSING" $rel
}
}

New-Item -ItemType Directory -Force -Path $state | Out-Null
foreach ($dir in @("runs","proposals","logs","receipts","locks")) {
New-Item -ItemType Directory -Force -Path (Join-Path $state $dir) | Out-Null
}

$interactiveLock = Join-Path $state "ACTIVE-INTERACTIVE.lock"
if (Test-Path $interactiveLock) { Fail "DREAM_DEFERRED" "ACTIVE-INTERACTIVE.lock present" }

$cycleLock = Join-Path $state "locks\dream-cycle.lock"
if (Test-Path $cycleLock) { Fail "BLOCKED_CONCURRENT_DREAM" $cycleLock }
Set-Content -LiteralPath $cycleLock -Value "$PID $(Get-Date -Format o)" -Encoding UTF8

try {
Push-Location $repo
try {
$head = (& git rev-parse HEAD).Trim()
if ($head -notmatch '^[a-f0-9]{40}$') { Fail "BLOCKED_HEAD_INVALID" $head }

    $now = Get-Date
    $parts = $NightStop.Split(':')
    $deadline = Get-Date -Hour ([int]$parts[0]) -Minute ([int]$parts[1]) -Second 0
    if ($deadline -le $now) { $deadline = $deadline.AddDays(1) }

    $runId = $now.ToString("yyyyMMdd-HHmmss")
    $proposal = Join-Path $state "proposals\dream-$runId.json"
    $library = Join-Path $state "pattern-library.json"
    $pass1Log = Join-Path $state "logs\dream-$runId-pass1.log"
    $pass2Log = Join-Path $state "logs\dream-$runId-pass2.log"
    $judgeLog = Join-Path $state "logs\dream-$runId-judge.json"
    $receiptPath = Join-Path $state "receipts\dream-$runId.json"
    $stopFlag = Join-Path $state "STOP.REQUESTED"

    if (Test-Path $stopFlag) { Remove-Item $stopFlag -Force }

    function Assert-Window() {
        if (Test-Path $stopFlag) { Fail "STOPPED_BY_REQUEST" "STOP.REQUESTED" }
        if ((Get-Date) -ge $deadline) { Fail "STOPPED_BY_DEADLINE" $deadline.ToString("o") }
    }

    Assert-Window

    $args = @(
        "scripts/memory-linker/dream-review-run.mjs",
        "--output", $proposal,
        "--head", $head,
        "--library", $library
    )
    if (-not [string]::IsNullOrWhiteSpace($InputJsonl)) {
        $inputFull = [System.IO.Path]::GetFullPath($InputJsonl)
        if (-not (Test-Path $inputFull -PathType Leaf)) { Fail "BLOCKED_INPUT_MISSING" $inputFull }
        $args += @("--input", $inputFull)
    }

    & node @args *> $pass1Log
    if ($LASTEXITCODE -ne 0) { Fail "NIGHT_BLOCKED_PASS1" "dream-review-run exit=$LASTEXITCODE" }
    Assert-Window

    $hash1 = (Get-FileHash -Algorithm SHA256 -LiteralPath $proposal).Hash.ToLowerInvariant()

    & node @args *> $pass2Log
    if ($LASTEXITCODE -ne 0) { Fail "NIGHT_BLOCKED_PASS2" "dream-review-run exit=$LASTEXITCODE" }
    Assert-Window

    $hash2 = (Get-FileHash -Algorithm SHA256 -LiteralPath $proposal).Hash.ToLowerInvariant()
    if ($hash1 -ne $hash2) {
        Fail "NIGHT_BLOCKED_NONDETERMINISTIC" "proposal sha mismatch $hash1 != $hash2"
    }

    $judge = Join-Path $PSScriptRoot "dream-house-judge.mjs"
    & node $judge $repo $proposal $head *> $judgeLog
    if ($LASTEXITCODE -ne 0) { Fail "NIGHT_BLOCKED_VALIDATION" "local judge exit=$LASTEXITCODE" }
    Assert-Window

    $tests = @(
        "scripts/memory-linker/dream-review.test.mjs",
        "scripts/memory-linker/pattern-discovery.test.mjs",
        "scripts/memory-linker/pattern-library.test.mjs",
        "scripts/memory-linker/memory-reserve-events.test.mjs",
        "scripts/memory-linker/dream-review-run.test.mjs"
    )
    $testResults = @()
    foreach ($test in $tests) {
        Assert-Window
        $testLog = Join-Path $state ("logs\test-" + [IO.Path]::GetFileNameWithoutExtension($test) + "-$runId.log")
        & node --test $test *> $testLog
        $ok = $LASTEXITCODE -eq 0
        $testResults += [ordered]@{ test=$test; pass=$ok; log=$testLog }
        if (-not $ok) { Fail "NIGHT_BLOCKED_REGRESSION" $test }
    }

    $proposalObj = Get-Content -Raw -LiteralPath $proposal | ConvertFrom-Json
    $receipt = [ordered]@{
        schema_version = 1
        mode = "HERMES_DREAM_HOUSE"
        run_id = $runId
        status = "NIGHT_PASS"
        pass = $true
        dry_run = [bool]$DryRun
        head_sha = $head
        started_at = $now.ToString("o")
        deadline = $deadline.ToString("o")
        proposal = $proposal
        proposal_sha256 = $hash2
        replay_stable = $true
        candidate_count = $proposalObj.candidate_count
        pattern_count = $proposalObj.pattern_discovery.pattern_count
        direct_apply = $false
        canonical_write = $false
        commit = $false
        push = $false
        merge = $false
        deploy = $false
        tests = $testResults
        judge_receipt = $judgeLog
    }
    $receipt | ConvertTo-Json -Depth 10 | Set-Content -LiteralPath $receiptPath -Encoding UTF8
    Copy-Item -LiteralPath $receiptPath -Destination (Join-Path $state "receipts\latest.json") -Force
    $receipt | ConvertTo-Json -Depth 10
} finally {
    Pop-Location
}
} finally {
Remove-Item -LiteralPath $cycleLock -Force -ErrorAction SilentlyContinue
}
""")

put("scripts/casa/install-dream-scheduler.ps1", r"""
[CmdletBinding()]
param(
[Parameter(Mandatory=$true)][string]$RepoRoot,
[Parameter(Mandatory=$true)][string]$StateRoot,
[Parameter(Mandatory=$true)][ValidatePattern('^\d{2}:\d{2}$')][string]$NightStart,
[Parameter(Mandatory=$true)][ValidatePattern('^\d{2}:\d{2}$')][string]$NightStop,
[Parameter(Mandatory=$true)][ValidatePattern('^\d{2}:\d{2}$')][string]$WakeTime,
[ValidateSet("WindowsTaskScheduler")][string]$SchedulerMode = "WindowsTaskScheduler",
[switch]$Apply
)

$repo = [System.IO.Path]::GetFullPath($RepoRoot)
$state = [System.IO.Path]::GetFullPath($StateRoot)

$wake = Join-Path $repo "scripts\casa\dream-house-wake.ps1"

repo_root = $repo
state_root = $state
night_start = $NightStart
night_stop = $NightStop
wake_time = $WakeTime
tasks = @(
"CentralDark-Hermes-Dream-Start",
"CentralDark-Hermes-Dream-Wake"
)
warning = "ONLY USE AFTER scheduler discovery proves no conflicting canonical owner."
}

$plan | ConvertTo-Json -Depth 8
exit 0
}

if (-not (Test-Path $cycle -PathType Leaf)) { throw "CYCLE_SCRIPT_MISSING=$cycle" }
if (-not (Test-Path $wake -PathType Leaf)) { throw "WAKE_SCRIPT_MISSING=$wake" }

$wakeArgs = "-NoProfile -ExecutionPolicy Bypass -File "$wake" -StateRoot "$state""

$startSettings = New-ScheduledTaskSettingsSet -StartWhenAvailable -MultipleInstances IgnoreNew

$wakeAction = New-ScheduledTaskAction -Execute $ps -Argument $wakeArgs

$wakeSettings = New-ScheduledTaskSettingsSet -StartWhenAvailable -MultipleInstances IgnoreNew

$plan.status = "INSTALLED"
$plan | ConvertTo-Json -Depth 8
""")
