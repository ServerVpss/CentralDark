<script>
  function fecharPopup() {
    document.getElementById('popup-termos').style.display = 'none';
    localStorage.setItem('ultimaExibicaoPopup', new Date().getTime());
  }

  function tocarSomPopup() {
    const audio = document.getElementById('popup-som');
    if (audio) {
      audio.play().catch(err => {
        // navegador pode bloquear autoplay — só exibe aviso no console
        console.warn("Som bloqueado pelo navegador até haver interação do usuário.", err);
      });
    }
  }

  window.addEventListener('load', () => {
    const agora = new Date().getTime();
    const ultimaExibicao = localStorage.getItem('ultimaExibicaoPopup');
    const intervalo = 10 * 60 * 1000; // 10 minutos

    if (!ultimaExibicao || agora - parseInt(ultimaExibicao) > intervalo) {
      const popup = document.getElementById('popup-termos');
      const box = popup.querySelector('.popup-box');
      popup.style.display = 'flex';

      void box.offsetWidth;
      box.classList.add('show');

      tocarSomPopup(); // Toca o som ao exibir
      setTimeout(() => fecharPopup(), 20000); // Fecha após 20 segundos
    }
  });
</script>
